DEFINE TABLE bug SCHEMAFULL;
DEFINE FIELD signature ON bug TYPE string;
DEFINE FIELD embedding ON bug TYPE array<float>;
DEFINE INDEX bug_vec ON bug FIELDS embedding
  HNSW DIMENSION 1024 DIST COSINE;

DEFINE TABLE fixed_by TYPE RELATION
  IN bug OUT fix;
DEFINE TABLE similar_to TYPE RELATION
  IN bug OUT bug;
DEFINE FIELD score ON similar_to TYPE float;

-- recall: nearest past bugs + their fixes
SELECT id, signature,
  ->fixed_by->fix.* AS fixes,
  vector::similarity::cosine(embedding, $e) AS sim
FROM bug WHERE embedding <|3|> $e;

-- dashboard stream
LIVE SELECT * FROM fixed_by, similar_to;

export type Stage =
  'detect' | 'triage' | 'recall' | 'reproduce' | 'patch'
  | 'verify' | 'gate' | 'deploy' | 'judge' | 'learn';

export type IncidentEvent =
  | { type: 'incident.opened'; id: string; flow: string; error: string }
  | { type: 'stage.started'; id: string; stage: Stage; sponsor: Sponsor }
  | { type: 'anchor.step'; id: string; step: number; ok: boolean; shot?: string }
  | { type: 'memory.match'; id: string; bugId: string; similarity: number }
  | { type: 'sandbox.log'; id: string; sandboxId: string; line: string }
  | { type: 'llm.token'; id: string; text: string }
  | { type: 'patch.ready'; id: string; diff: string; confidence: number }
  | { type: 'deploy.progress'; id: string; percent: 5 | 25 | 100; healthy: boolean }
  | { type: 'eval.scored'; id: string; score: number; traceUrl: string }
  | { type: 'incident.closed'; id: string; outcome: 'healed' | 'rolled_back' | 'escalated' };

export type Sponsor = 'anchor' | 'surreal' | 'daytona' | 'bedrock' | 'aws' | 'arize';

// packages/agent/stages/reproduce.ts
export const reproduce: Stage = async (inc, ctx) => {
  const sb = await daytona.call({
    repo: inc.repo, sha: inc.sha,
  }, ctx);
  const res = await sb.run(
    'npx playwright test ' + inc.flow.spec);

  ctx.emit({ type: 'sandbox.log', ... });
  if (res.exitCode === 0)
    return { next: 'triage', note: 'not reproducible' };
  return { next: 'patch', evidence: res.output };
};
// packages/agent/machine.ts
export async function drive(inc: Incident) {
  let stage: Stage = inc.stage;
  while (stage !== 'closed') {
    const out = await tracer.startActiveSpan(stage,
      (span) => STAGES[stage](inc, ctxFor(inc, span)));
    await memory.transition(inc.id, stage, out);
    emit({ type: 'stage.started', id: inc.id,
           stage: out.next, sponsor: SPONSOR_OF[out.next] });
    stage = out.next;
  }
}
